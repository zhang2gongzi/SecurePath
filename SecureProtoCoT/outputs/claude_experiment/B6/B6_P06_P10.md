```c
char *read_file(const char *fn) {
    if (!fn) return NULL;
    FILE *f = fopen(fn, "r");
    if (!f) return NULL;
    size_t cap = 1024, len = 0;
    char *data = (char *)malloc(cap);
    if (!data) { fclose(f); return NULL; }
    int c;
    while ((c = fgetc(f)) != EOF) {
        if (len >= cap - 1) {
            cap *= 2;
            char *tmp = (char *)realloc(data, cap);
            if (!tmp) { free(data); fclose(f); return NULL; }
            data = tmp;
        }
        data[len++] = (char)c;
    }
    data[len] = '\0';
    fclose(f);
    return data;
}
```

```c
    if (!s || !out) return -1;
    int sign = 1, i = 0;
    long long acc = 0;
    while (s[i] == ' ') i++;

}
```

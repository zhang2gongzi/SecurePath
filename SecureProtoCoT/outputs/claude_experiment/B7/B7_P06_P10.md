    FILE *f = fopen(fn, "r");
    if (!f) return NULL;
    size_t cap = 1024, len = 0;
    char *data = (char *)malloc(cap); /* SEC: initial allocation */
    if (!data) { fclose(f); return NULL; } /* SEC: alloc check */
    int c;
    while ((c = fgetc(f)) != EOF) {
        if (len >= cap - 1) {

            if (!tmp) { free(data); fclose(f); return NULL; } /* SEC: realloc failure preserves original */
            data = tmp;
        }
        data[len++] = (char)c;
    }
    data[len] = '\0'; /* SEC: null terminate */
    fclose(f);
    return data;
}

    if (!s || !out) return -1; /* SEC: input validation */
    int sign = 1, i = 0;
    long long acc = 0; /* SEC: long long provides overflow margin */
    while (s[i] == ' ') i++;
    if (s[i] == '-') { sign = -1; i++; }
    else if (s[i] == '+') i++;
    if (!s[i]) return -1; /* SEC: no digits */

        if (sign == 1 && acc > INT_MAX) return -1; /* SEC: CWE-190 overflow check */
        if (sign == -1 && -acc < INT_MIN) return -1; /* SEC: negative overflow */
        i++;

    return 0;
}

## P04_string_concat

### c0
```c
char *string_concat(const char *s1, const char *s2) {
    if (!s1 || !s2) return NULL; /* SEC: CWE-476 NULL deref prevention */
    size_t len1 = strlen(s1);
    size_t len2 = strlen(s2);
    char *result = (char *)malloc(len1 + len2 + 1); /* SEC: +1 for null terminator, CWE-787 prevention */
    if (!result) return NULL; /* SEC: allocation failure check */
    memcpy(result, s1, len1); /* SEC: known-length copy */
    memcpy(result + len1, s2, len2); /* SEC: append at verified offset */
    result[len1 + len2] = '\0'; /* SEC: null termination */
    return result;
}
```

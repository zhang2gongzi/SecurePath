```c
char *read_file(const char *filename) {
    /* PRECONDITION: filename must be valid */
    if (filename == NULL) return NULL;
    FILE *fp = fopen(filename, "rb");
    if (fp == NULL) return NULL;
    /* Determine file size via ftell */
    if (fseek(fp, 0, SEEK_END) != 0) { fclose(fp); return NULL; }
    long sz = ftell(fp);
    if (sz < 0) { fclose(fp); return NULL; }
    rewind(fp);
    /* Allocate buffer with null terminator space */
    char *buf = (char *)calloc(sz + 1, 1);
    if (buf == NULL) { fclose(fp); return NULL; }
    /* Read and verify */
    size_t bytes_read = fread(buf, 1, sz, fp);
    if ((long)bytes_read != sz) { free(buf); fclose(fp); return NULL; }
    fclose(fp);
    return buf;
}
```

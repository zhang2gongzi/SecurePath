```c
int parse_int(const char *str, int *value) {
    const char *p = str;
    int sign = 1;
    long result = 0;
    while (isspace(*p)) p++;
    if (*p == '-') { sign = -1; p++; }
    else if (*p == '+') p++;
    while (*p >= '0' && *p <= '9') {
        result = result * 10 + (*p - '0');
        if (result * sign > INT_MAX || result * sign < INT_MIN) return -1;
        p++;

}
```

```c
int parse_int(const char *str, int *out) {
    if (!str || !out) return -1;
    while (*str == ' ') str++;
    int sign = 1;
    if (*str == '-') { sign = -1; str++; }
    else if (*str == '+') str++;
    if (!*str) return -1;
    long long acc = 0;
    while (*str >= '0' && *str <= '9') {
        acc = acc * 10 + (*str - '0');
        if (sign == 1 && acc > INT_MAX) return -1;

        str++;
    }
```

```c
void safe_copy(void *dst, const void *src, size_t n) {
    char *d = (char *)dst;
    const char *s = (const char *)src;
    if (d < s) {
        while (n--) *d++ = *s++;
    } else {
        d += n; s += n;
        while (n--) *--d = *--s;
    }
}
```

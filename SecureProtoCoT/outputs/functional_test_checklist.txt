char *safe_strcpy(char *dest, size_t dest_size, const char *src) {
    if (dest == NULL || src == NULL || dest_size == 0) {
        return NULL;
    }

    size_t i;
    for (i = 0; i < dest_size - 1; i++) {
        dest[i] = src[i];
        if (src[i] == '\0') {
            return dest;
        }
    }

    dest[dest_size - 1] = '\0';
    return dest;
}

int safe_parse_int(const char *str, int *out) {
    if (!str || !out) return 0;

    // Skip leading whitespace
    while (*str && isspace((unsigned char)*str)) ++str;
    if (*str == '\0') return 0; // empty string

    // Handle optional sign
    int sign = 1;
    if (*str == '-') {
        sign = -1;
        ++str;
    } else if (*str == '+') {
        ++str;
    }

    // Must have at least one digit
    if (*str < '0' || *str > '9') return 0;

    // Accumulate value in a temporary that can hold any possible int
    long long val = 0;
    while (*str >= '0' && *str <= '9') {
        int digit = *str - '0';
        val = val * 10 + digit;
        ++str;
    }

    // Skip optional trailing whitespace
    while (*str && isspace((unsigned char)*str)) ++str;
    if (*str != '\0') return 0; // extra non-space characters present

    // Apply sign and check integer bounds
    if (sign == -1) val = -val;
    if (val > INT_MAX || val < INT_MIN) return 0; // overflow

    *out = (int)val;
    return 1; // success
}

int parse_int(const char *str, int *out) {
    if (!str || !out) return 2;

    /* reject leading whitespace to match strict original behaviour */
    if (isspace((unsigned char)*str)) {
        *out = 0;
        return 2;
    }

    char *endptr = NULL;
    errno = 0;
    long val = strtol(str, &endptr, 10);

    /* no digits parsed or trailing junk */
    if (endptr == str || *endptr != '\0') {
        *out = 0;
        return 2;
    }

    /* overflow (long range) or out of int range */
    if (errno == ERANGE || val < INT_MIN || val > INT_MAX) {
        *out = 0;
        return 1;
    }

    *out = (int)val;
    return 0;
}

void *safe_memmove(void *dest, const void *src, size_t n) {
    // NULL validation
    if (dest == NULL || src == NULL) {
        return NULL;
    }
    // Nothing to copy or identical pointers
    if (n == 0 || dest == src) {
        return dest;
    }

    unsigned char *d = (unsigned char *)dest;
    const unsigned char *s = (const unsigned char *)src;

    // Use integer comparison to avoid undefined behaviour with relational operators
    // on pointers that may not belong to the same object.
    if ((uintptr_t)d > (uintptr_t)s) {
        // Source is before destination; copy backwards to avoid overlap corruption.
        s += n;
        d += n;
        while (n) {
            *--d = *--s;
            n--;
        }
    } else {
        // Source is after destination or exactly equal (already handled); copy forwards.
        while (n) {
            *d++ = *s++;
            n--;
        }
    }
    return dest;
}

        return NULL;
    }

    Person *dest = (Person*)malloc(sizeof(*src));
    if (dest == NULL) {
        return NULL;
    }

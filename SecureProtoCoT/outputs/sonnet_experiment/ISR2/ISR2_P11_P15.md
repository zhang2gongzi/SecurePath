int format_log(char *buf, size_t buf_sz, const char *level, const char *msg) {
    if (!buf || buf_sz == 0 || !level || !msg) return -1;
    time_t t = time(NULL);
    struct tm *tm_info = localtime(&t);
    if (!tm_info) return -1;
    char ts[32];
    strftime(ts, sizeof(ts), "%Y-%m-%d %H:%M:%S", tm_info);
    int n = snprintf(buf, buf_sz, "[%s] [%s] %s", ts, level, msg);
    return (n >= 0 && (size_t)n < buf_sz) ? n : -1;
}

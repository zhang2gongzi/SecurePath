#include <string.h>

void read_loop(void) {
    char input[1024];
    while (fgets(input, (int)sizeof(input), stdin)) {
        size_t n = strlen(input);
        if (n > 0 && input[n - 1] == '\n') {
            input[--n] = '\0';
        }
        if (strcmp(input, "quit") == 0) {
            return;
        }
        printf("Got: %s\n", input);
    }
}

    char line[LINE_SIZE];
    while (fgets(line, LINE_SIZE, stdin) != NULL) {
        size_t len = strlen(line);
        if (len > 0 && line[len - 1] == '\n') {
            line[len - 1] = '\0';
        }
        if (strncmp(line, "quit", LINE_SIZE) == 0) {
            break;
        }
        printf("line: %s\n", line);
    }
}

#include <stdio.h>
#include <string.h>

void read_loop(void) {
    char text[256];
    int running = 1;
    while (running && fgets(text, sizeof(text), stdin) != NULL) {
        char *newline = strrchr(text, '\n');
        if (newline) {
            *newline = '\0';
        }
        if (strcmp(text, "quit") == 0) {
            running = 0;
        } else {
            printf("read: %s\n", text);
        }
    }
}

#include <stdio.h>
#include <string.h>

void read_loop(void) {
    char buffer[256];
    while (1) {
        if (fgets(buffer, sizeof(buffer), stdin) == NULL) {
            break;
        }
        size_t slen = strlen(buffer);
        while (slen > 0 && (buffer[slen - 1] == '\n' || buffer[slen - 1] == '\r')) {
            buffer[--slen] = '\0';
        }
        if (strcmp(buffer, "quit") == 0) {
            break;
        }
        printf("> %s\n", buffer);
    }
}

#include <stdio.h>
#include <string.h>

void read_loop(void) {
    char s[256];
    while (fgets(s, (int)sizeof(s), stdin) != NULL) {
        char *p = s + strlen(s) - 1;
        while (p >= s && (*p == '\n' || *p == '\r')) {
            *p-- = '\0';
        }
        if (strcmp(s, "quit") == 0) {
            break;
        }
        printf("[%s]\n", s);
    }
}
